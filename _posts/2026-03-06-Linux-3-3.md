# 사용자 및 파일 권한 관리 — 3-3 실습편 (초보자 버전)


---

## Step 0. 터미널 시작 확인

```bash
whoami                 # 지금 로그인한 사용자 이름 확인
echo $SHELL            # 현재 로그인 셸 확인 (예: /bin/bash)
pwd                    # 현재 폴더 위치 확인
```

---

## Step 1. 실습 폴더 만들기

```bash
mkdir ~/linux-journey                # 실습 폴더 생성
mkdir ~/linux-journey/docs           # 문서 폴더 생성
mkdir ~/linux-journey/tmp            # 임시 폴더 생성
mkdir ~/linux-journey/bin            # 스크립트 폴더 생성

cd ~/linux-journey                   # 실습 폴더로 이동
pwd                                  # 이동 확인
ls                                   # 현재 폴더 목록 확인
```

---

## Step 2. 파일 기본 명령 익히기

```bash
touch docs/intro.txt                 # 빈 파일 생성
echo "Linux Fresh Install Lab" > docs/intro.txt   # 파일에 한 줄 저장
cat docs/intro.txt                   # 파일 내용 확인

cp docs/intro.txt docs/intro_copy.txt   # 파일 복사
mv docs/intro_copy.txt docs/intro_moved.txt  # 파일 이름 변경
ls -l docs                           # docs 폴더 상세 목록 확인

touch tmp/remove_me.txt              # 삭제 연습용 파일 생성
rm tmp/remove_me.txt                 # 파일 삭제
ls tmp                               # 삭제되었는지 확인
```

---

## Step 3. 링크(soft/hard) 아주 쉽게 확인

```bash
echo "link-test" > docs/original.txt    # 원본 파일 만들기

ln -s docs/original.txt docs/soft_link.txt   # 심볼릭(소프트) 링크 만들기
ln docs/original.txt docs/hard_link.txt      # 하드 링크 만들기

ls -l docs/original.txt               # 원본 확인
ls -l docs/soft_link.txt              # 소프트 링크 확인
ls -l docs/hard_link.txt              # 하드 링크 확인

ls -i docs/original.txt               # 원본 inode 확인
ls -i docs/soft_link.txt              # 소프트 링크 inode 확인
ls -i docs/hard_link.txt              # 하드 링크 inode 확인

rm docs/original.txt                  # 원본 삭제
cat docs/soft_link.txt                # 소프트 링크 확인(보통 실패)
cat docs/hard_link.txt                # 하드 링크 확인(보통 성공)
```

---

## Step 4. 아주 간단한 스크립트 실행

```bash
nano bin/week3_bootstrap.sh
```

아래 내용을 입력:

```bash
#!/bin/bash
echo "=== Week3 Bootstrap ==="
echo "USER: $(whoami)"
echo "SHELL: $SHELL"
echo "PWD: $(pwd)"
date
```

저장 후 실행:

```bash
chmod +x bin/week3_bootstrap.sh       # 실행 권한 추가
./bin/week3_bootstrap.sh              # 스크립트 실행
```

## Step 5. 사용자/그룹 만들기 (3-1 연결)

```bash
sudo adduser pig                      # 사용자 pig 생성
sudo adduser mango                    # 사용자 mango 생성
sudo adduser dog                      # 사용자 dog 생성

sudo addgroup animals                 # 그룹 animals 생성
sudo adduser pig animals              # pig를 animals 그룹에 추가
sudo adduser mango animals            # mango를 animals 그룹에 추가
sudo adduser dog animals              # dog를 animals 그룹에 추가

id pig                                # pig 계정 정보 확인
id mango                              # mango 계정 정보 확인
id dog                                # dog 계정 정보 확인

grep pig /etc/passwd                  # pig 사용자 레코드 확인
grep mango /etc/passwd                # mango 사용자 레코드 확인
grep dog /etc/passwd                  # dog 사용자 레코드 확인
grep animals /etc/group               # animals 그룹 레코드 확인
```

---

## Step 6. 소유권/권한 적용 (3-2 연결)

```bash
sudo mkdir /tmp/lab-perm                      # 실습 디렉터리 생성
sudo chown root:animals /tmp/lab-perm         # 소유자/그룹 설정
sudo chmod 775 /tmp/lab-perm                  # 디렉터리 권한 설정

su - dog                                       # dog 사용자로 전환
echo "dog baseline" > /tmp/lab-perm/from_dog.txt   # dog 파일 생성
exit                                           # 다시 원래 사용자로 복귀

sudo chown dog:animals /tmp/lab-perm/from_dog.txt    # 파일 그룹을 animals로 맞춤
sudo chmod 640 /tmp/lab-perm/from_dog.txt            # 파일 권한 제한

ls -ld /tmp/lab-perm                                # 디렉터리 상태 확인
ls -l /tmp/lab-perm/from_dog.txt                    # 파일 상태 확인

su - mango                                          # mango로 전환
cat /tmp/lab-perm/from_dog.txt                      # 읽기 테스트(성공 기대)
exit
```

---

## Step 7. Sticky bit / SGID / SUID 확인

```bash
sudo mkdir /tmp/sticky-lab                         # sticky 실습 폴더 생성
sudo chmod 777 /tmp/sticky-lab                     # 모두 쓰기 가능

su - pig                                            # pig로 전환
echo "pig file" > /tmp/sticky-lab/pig.txt         # pig 파일 생성
exit

su - mango                                          # mango로 전환
rm /tmp/sticky-lab/pig.txt                          # sticky 없을 때 삭제 가능
exit

su - pig                                            # pig로 재생성
echo "pig file 2" > /tmp/sticky-lab/pig.txt
exit

sudo chmod 1777 /tmp/sticky-lab                    # sticky bit 적용

su - mango                                          # mango로 재시도
rm /tmp/sticky-lab/pig.txt                          # 이번에는 실패가 정상
exit

ls -ld /tmp/sticky-lab                              # drwxrwxrwt 확인
ls -l /tmp/sticky-lab                               # pig.txt 남아있는지 확인
```

```bash
sudo mkdir /tmp/team-share                          # SGID 실습 폴더 생성
sudo chown root:animals /tmp/team-share             # 그룹 animals
sudo chmod 2775 /tmp/team-share                     # SGID 적용

su - pig
echo "pig note" > /tmp/team-share/pig_note.txt
exit

su - mango
echo "mango note" > /tmp/team-share/mango_note.txt
exit

ls -ld /tmp/team-share                              # drwxrwsr-x 확인
ls -l /tmp/team-share                               # 파일 그룹 상속 확인

ls -l /usr/bin/passwd                               # SUID 예시 확인
stat /usr/bin/passwd                                # 상세 메타데이터 확인
```

---

## 완료 확인

```bash
id pig
id mango
id dog

ls -ld /tmp/lab-perm
ls -ld /tmp/sticky-lab
ls -ld /tmp/team-share
```

완료 기준

- `mango`가 `/tmp/lab-perm/from_dog.txt`를 읽을 수 있다.
- `sticky-lab`는 `t`가 보이고, `mango`가 `pig.txt`를 지우지 못한다.
- `team-share`는 `s`가 보이고, 생성 파일 그룹이 `animals`다.
