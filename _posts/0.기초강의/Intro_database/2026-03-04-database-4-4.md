# 4-4 input()으로 검색하기

사용자가 입력한 값으로 상품을 검색한다.

---

## 1. 파일 만들기

파일명: `step4_4_input_search.py`

```python
import mysql.connector

# mysql.connector.connect(...)
# - MySQL 서버에 접속한다.
conn = mysql.connector.connect(
    host="localhost",       # DB 서버 주소
    user="root",            # DB 계정
    password="비밀번호",    # 본인 비밀번호로 변경
    database="datatype_db", # 사용할 DB
    charset="utf8mb4"       # 한글 처리 문자셋
)

# conn.cursor()
# - SQL 실행 도구(cursor)를 만든다.
cursor = conn.cursor()

# 메뉴를 화면에 출력한다.
print("=== 상품 검색 프로그램 ===")
print("1. 이름 검색 (LIKE)")
print("2. 가격 범위 검색 (BETWEEN)")

# input(...).strip()
# - 사용자 입력을 문자열로 받고, 앞뒤 공백을 제거한다.
menu = input("번호 입력 (1 또는 2): ").strip()

# 메뉴 1: 이름 검색
if menu == "1":
    # 검색어를 입력받는다.
    keyword = input("검색어 입력: ").strip()

    # %s 바인딩 자리를 사용한 SQL
    sql = "SELECT id, name, price FROM products WHERE name LIKE %s"

    # 포함 검색을 위해 %검색어% 형태로 만든다.
    value = (f"%{keyword}%",)

    # cursor.execute(sql, value)
    # - SQL 실행 + 값 바인딩
    cursor.execute(sql, value)

    # cursor.fetchall()
    # - 조회 결과를 리스트로 가져온다.
    rows = cursor.fetchall()

    print(f"\\n[이름 검색 결과: {keyword}]")
    if rows:
        # 결과가 있으면 한 줄씩 출력
        for row in rows:
            print(row)
    else:
        # 결과가 없으면 안내 문구 출력
        print("검색 결과가 없습니다.")

# 메뉴 2: 가격 범위 검색
elif menu == "2":
    min_price = input("최소 가격: ").strip()
    max_price = input("최대 가격: ").strip()

    # isdigit()
    # - 숫자 입력인지 검증한다.
    if not min_price.isdigit() or not max_price.isdigit():
        print("가격은 숫자로 입력해야 합니다.")
    else:
        sql = "SELECT id, name, price FROM products WHERE price BETWEEN %s AND %s"
        value = (int(min_price), int(max_price))

        cursor.execute(sql, value)
        rows = cursor.fetchall()

        print(f"\\n[가격 검색 결과: {min_price} ~ {max_price}]")
        if rows:
            for row in rows:
                print(row)
        else:
            print("검색 결과가 없습니다.")

# 잘못된 메뉴 입력 처리
else:
    print("1 또는 2만 입력하세요.")

# cursor.close(), conn.close()
# - 사용한 자원을 정리한다.
cursor.close()
conn.close()
print("\\n프로그램 종료")
```
