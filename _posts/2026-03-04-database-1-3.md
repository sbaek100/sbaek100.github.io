## 2. venv 가상환경 설정

### venv란?

가상환경(Virtual Environment)은 프로젝트별로 독립된 Python 패키지 공간을 만드는 도구임.  
시스템 전역 Python 환경과 분리되어 패키지 버전 충돌을 방지함.

```
시스템 Python
├── 프로젝트A/venv  ← mysql-connector 2.0
└── 프로젝트B/venv  ← mysql-connector 3.0  (충돌 없음)
```

### 가상환경 생성 및 활성화

```bash
# 1. 프로젝트 폴더 생성
mkdir coffee_db
cd coffee_db

# 2. 가상환경 생성 (.venv 폴더가 생성됨)
python -m venv .venv

# 3. 가상환경 활성화 (Windows)
.venv\Scripts\activate

# 3. 가상환경 활성화 (macOS/Linux)
source .venv/bin/activate

# 활성화 성공 시 프롬프트 앞에 (.venv) 표시됨
# (.venv) C:\coffee_db>
```

### 가상환경 비활성화

```bash
deactivate
```

> 💡 **[Tip]** `.venv` 폴더는 Git에 올리지 않음. `.gitignore`에 `.venv/` 추가 권장.

## 6. 전체 CRUD 코드

### 6-1. DB 및 테이블 초기화

```python
# db_setup.py
import mysql.connector
from mysql.connector import Error

def setup_database():
    """DB 생성, 테이블 생성, 초기 데이터 삽입"""

    # 1단계: DB 없이 서버에만 접속 (DB 생성 전이므로 database 파라미터 제외)
    try:
        conn = mysql.connector.connect(
            host="localhost",
            user="root",
            password="your_password",
            charset="utf8mb4"
        )
        cursor = conn.cursor()

        # DB 생성
        cursor.execute("CREATE DATABASE IF NOT EXISTS starbuuks")
        print("✅ starbuuks DB 생성 완료")

        # DB 선택
        cursor.execute("USE starbuuks")

        # 테이블 생성
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS coffees (
                id    INT         NOT NULL AUTO_INCREMENT,
                name  VARCHAR(20) NOT NULL,
                price INT         NOT NULL,
                PRIMARY KEY (id)
            )
        """)
        print("✅ coffees 테이블 생성 완료")

        # 초기 데이터 삽입 (이미 데이터가 있으면 건너뜀)
        cursor.execute("SELECT COUNT(*) FROM coffees")
        count = cursor.fetchone()[0]

        if count == 0:
            initial_data = [
                ("아메리카노", 3800),
                ("카페라떼",   4000),
                ("콜드브루",   3500),
                ("카페모카",   4500),
                ("카푸치노",   5000),
            ]
            cursor.executemany(
                "INSERT INTO coffees (name, price) VALUES (%s, %s)",
                initial_data
            )
            conn.commit()  # 변경사항 확정 (INSERT/UPDATE/DELETE는 반드시 commit 필요)
            print(f"✅ 초기 데이터 {len(initial_data)}건 삽입 완료")
        else:
            print(f"ℹ️  이미 데이터 {count}건 존재 — 삽입 건너뜀")

    except Error as e:
        print(f"❌ DB 오류: {e}")

    finally:
        # 연결은 항상 닫아줘야 함
        if cursor:
            cursor.close()
        if conn:
            conn.close()

if __name__ == "__main__":
    setup_database()
```

### 6-2. CRUD 함수 모음 (main.py)

```python
# main.py
import mysql.connector
from mysql.connector import Error
from config import DB_CONFIG


# ─────────────────────────────────────────
# 연결 헬퍼 함수
# ─────────────────────────────────────────
def get_connection():
    """DB 연결 객체 반환"""
    return mysql.connector.connect(**DB_CONFIG)


# ─────────────────────────────────────────
# READ: 전체 조회
# ─────────────────────────────────────────
def select_all():
    """coffees 테이블 전체 조회"""
    try:
        conn = get_connection()
        cursor = conn.cursor()

        cursor.execute("SELECT * FROM coffees")
        rows = cursor.fetchall()  # 결과 전체를 리스트로 반환

        print("\n 전체 커피 목록")
        print(f"{'ID':<5} {'이름':<12} {'가격':>6}")
        print("-" * 25)
        for row in rows:
            print(f"{row[0]:<5} {row[1]:<12} {row[2]:>6}원")

    except Error as e:
        print(f"❌ 조회 오류: {e}")
    finally:
        cursor.close()
        conn.close()


# ─────────────────────────────────────────
# READ: 이름으로 특정 커피 조회
# ─────────────────────────────────────────
def select_by_name(name: str):
    """이름으로 커피 조회"""
    try:
        conn = get_connection()
        cursor = conn.cursor()

        # %s 플레이스홀더 사용 → SQL 인젝션(Injection) 방지 (Fact)
        cursor.execute(
            "SELECT * FROM coffees WHERE name = %s",
            (name,)   # 튜플 형태로 전달해야 함
        )
        row = cursor.fetchone()  # 단일 행 반환

        if row:
            print(f"\n 조회 결과: ID={row[0]}, 이름={row[1]}, 가격={row[2]}원")
        else:
            print(f"\n  '{name}' 커피를 찾을 수 없음")

    except Error as e:
        print(f" 조회 오류: {e}")
    finally:
        cursor.close()
        conn.close()


# ─────────────────────────────────────────
# CREATE: 새 커피 추가
# ─────────────────────────────────────────
def insert_coffee(name: str, price: int):
    """새 커피 데이터 삽입"""
    try:
        conn = get_connection()
        cursor = conn.cursor()

        cursor.execute(
            "INSERT INTO coffees (name, price) VALUES (%s, %s)",
            (name, price)
        )
        conn.commit()  # INSERT는 반드시 commit

        print(f" 추가 완료: '{name}' {price}원 (ID: {cursor.lastrowid})")

    except Error as e:
        print(f" 삽입 오류: {e}")
    finally:
        cursor.close()
        conn.close()


# ─────────────────────────────────────────
# UPDATE: 가격 수정
# ─────────────────────────────────────────
def update_price(name: str, new_price: int):
    """특정 커피의 가격 수정"""
    try:
        conn = get_connection()
        cursor = conn.cursor()

        cursor.execute(
            "UPDATE coffees SET price = %s WHERE name = %s",
            (new_price, name)
        )
        conn.commit()

        # rowcount: 영향받은 행 수 (0이면 해당 이름 없음)
        if cursor.rowcount > 0:
            print(f" 수정 완료: '{name}' 가격 → {new_price}원")
        else:
            print(f"  '{name}' 커피를 찾을 수 없음")

    except Error as e:
        print(f" 수정 오류: {e}")
    finally:
        cursor.close()
        conn.close()


# ─────────────────────────────────────────
# UPDATE: 가격 인상 (기존 가격 + 인상액)
# ─────────────────────────────────────────
def increase_price(name: str, amount: int):
    """특정 커피 가격 인상"""
    try:
        conn = get_connection()
        cursor = conn.cursor()

        cursor.execute(
            "UPDATE coffees SET price = price + %s WHERE name = %s",
            (amount, name)
        )
        conn.commit()

        if cursor.rowcount > 0:
            print(f" 가격 인상 완료: '{name}' +{amount}원")
        else:
            print(f"  '{name}' 커피를 찾을 수 없음")

    except Error as e:
        print(f" 수정 오류: {e}")
    finally:
        cursor.close()
        conn.close()


# ─────────────────────────────────────────
# DELETE: 커피 삭제
# ─────────────────────────────────────────
def delete_coffee(name: str):
    """특정 커피 삭제"""
    try:
        conn = get_connection()
        cursor = conn.cursor()

        cursor.execute(
            "DELETE FROM coffees WHERE name = %s",
            (name,)
        )
        conn.commit()

        if cursor.rowcount > 0:
            print(f" 삭제 완료: '{name}'")
        else:
            print(f"  '{name}' 커피를 찾을 수 없음")

    except Error as e:
        print(f" 삭제 오류: {e}")
    finally:
        cursor.close()
        conn.close()


# ─────────────────────────────────────────
# 셀프체크 문제 순서대로 실행
# ─────────────────────────────────────────
if __name__ == "__main__":

    print("=" * 40)
    print("  coffees 테이블 CRUD 실습")
    print("=" * 40)

    # 문제 4: 커피 이름만 조회
    print("\n[문제 4] 모든 커피 이름 조회")
    conn = get_connection()
    cursor = conn.cursor()
    cursor.execute("SELECT name FROM coffees")
    names = cursor.fetchall()
    for n in names:
        print(f"  - {n[0]}")
    cursor.close()
    conn.close()

    # 문제 5: 카푸치노 200원 인상
    print("\n[문제 5] 카푸치노 200원 인상")
    increase_price("카푸치노", 200)

    # 문제 6: 콜드브루 삭제
    print("\n[문제 6] 콜드브루 삭제")
    delete_coffee("콜드브루")

    # 문제 7: 전체 조회
    print("\n[문제 7] 전체 데이터 조회")
    select_all()
```

## 핵심 개념 요약

|개념|설명|비고|
|---|---|---|
|`mysql.connector.connect()`|MySQL 서버에 연결|`**DB_CONFIG` 딕셔너리 언패킹 사용|
|`cursor()`|SQL 실행 커서(Cursor) 생성|쿼리 실행의 실행 주체|
|`cursor.execute(sql, params)`|SQL 실행. params는 튜플|`%s` 플레이스홀더로 인젝션 방지|
|`cursor.fetchall()`|전체 결과 리스트 반환|SELECT 결과 전체|
|`cursor.fetchone()`|단일 행 반환|특정 조건 1건 조회|
|`conn.commit()`|변경사항 DB에 확정|INSERT/UPDATE/DELETE 후 필수|
|`cursor.rowcount`|영향받은 행 수|UPDATE/DELETE 성공 여부 확인|
|`conn.close()`|연결 종료|항상 finally 블록에서 호출|

> 💡 **[Opinion]** Python에서 SQL 파라미터를 전달할 때 문자열 포맷팅(`f"...{변수}..."`)을 절대 사용하지 말 것. 반드시 `%s` 플레이스홀더를 사용해야 **SQL 인젝션(Injection) 공격**을 방지할 수 있음. 이는 보안의 기본 원칙임.
