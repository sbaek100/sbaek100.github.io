# Python으로 MySQL 연동하기

## DB 생성 -> 기본 데이터 삽입 -> 사용자 입력 CRUD 실습

> 실습 흐름: DB/테이블 생성 -> 기본 데이터 삽입 -> 메뉴 기반 CRUD 반복  
> 환경: Windows / Python 3.10 이상 / MySQL 설치 완료 상태  
> 각 단계마다 코드를 작성하고, 실행하여 결과를 직접 확인한 후 다음 단계로 진행함.

## 6. [Step 2] DB 및 테이블 생성

연결이 확인되면 `starbuuks` DB와 `coffees` 테이블을 생성함.  
파일명: `step2_create_db.py`

```python
# step2_create_db.py - DB 및 테이블 생성
import mysql.connector

conn = mysql.connector.connect(
    host="localhost",
    user="root",
    password="your_password",
    charset="utf8mb4"
)
cursor = conn.cursor()

# DB 생성 (이미 있으면 그대로 유지)
cursor.execute(
    "CREATE DATABASE IF NOT EXISTS starbuuks DEFAULT CHARACTER SET utf8mb4"
)
print("DB 'starbuuks' 생성 완료 (이미 있으면 유지)")

# DB 선택
cursor.execute("USE starbuuks")

# 테이블 생성 (이미 있으면 그대로 유지)
cursor.execute("""
    CREATE TABLE IF NOT EXISTS coffees (
        id    INT AUTO_INCREMENT PRIMARY KEY,
        name  VARCHAR(50) NOT NULL,
        price INT         NOT NULL
    )
""")
print("테이블 'coffees' 생성 완료 (이미 있으면 유지)")

cursor.close()
conn.close()
print("완료")
```

## 7. [Step 3] 기본 데이터 삽입

```python
# step3_insert.py - 기본 데이터 삽입
import mysql.connector

    host="localhost",
    user="root",
    password="your_password",
    database="starbuuks",
    charset="utf8mb4"
)
cursor = conn.cursor()

# 현재 데이터 수 확인
cursor.execute("SELECT COUNT(*) FROM coffees")
count = cursor.fetchone()[0]

if count == 0:
    default_data = [
        ("아메리카노", 3800),
        ("카페라떼",   4000),
        ("카푸치노",   4500),
        ("카페모카",   4800),
        ("바닐라라떼", 5000),
    ]
    cursor.executemany(
        "INSERT INTO coffees (name, price) VALUES (%s, %s)",
        default_data
    )
    conn.commit()
    print(f"기본 데이터 {cursor.rowcount}건 삽입 완료")
else:
    print(f"기존 데이터 {count}건 존재 -> 삽입 건너뜀")

# 현재 테이블 전체 출력
cursor.execute("SELECT * FROM coffees ORDER BY id")
rows = cursor.fetchall()
print()
print(f"  {'ID':<5} {'이름':<12} {'가격':>6}")
print("  " + "-" * 26)
for row in rows:
    print(f"  {row[0]:<5} {row[1]:<12} {row[2]:>6}원")

cursor.close()
conn.close()
print("\n완료")
```

## 9. [Step 5] CREATE - 사용자 입력으로 추가

`input()`으로 이름과 가격을 입력받아 새 행을 삽입하는 함수를 만들고 테스트함.  
파일명: `step5_create.py`

```python
# step5_create.py - 사용자 입력으로 데이터 추가
import mysql.connector

def get_conn():
    return mysql.connector.connect(
        host="localhost",
        user="root",
        password="your_password",
        database="starbuuks",
        charset="utf8mb4"
    )

def show_all(cursor):
    cursor.execute("SELECT * FROM coffees ORDER BY id")
    rows = cursor.fetchall()
    print()
    print(f"  {'ID':<5} {'이름':<12} {'가격':>6}")
    print("  " + "-" * 26)
    for row in rows:
        print(f"  {row[0]:<5} {row[1]:<12} {row[2]:>6}원")
    print()

def create(conn, cursor):
    print("\n[CREATE] 새 커피 정보를 입력하세요.")
    name  = input("  커피 이름: ").strip()
    price = input("  가격 (원): ").strip()

        print("  입력값이 올바르지 않습니다. 이름은 필수, 가격은 숫자여야 합니다.")
        return

        "INSERT INTO coffees (name, price) VALUES (%s, %s)",
        (name, int(price))
    )
    conn.commit()
    print(f"  '{name}' 추가 완료 (ID: {cursor.lastrowid})")

# 단독 실행 테스트
conn   = get_conn()
cursor = conn.cursor()

print("[추가 전] 현재 목록:")
show_all(cursor)

create(conn, cursor)

print("[추가 후] 현재 목록:")
```

## 10. [Step 6] UPDATE - 사용자 입력으로 수정

ID를 입력받아 해당 행의 이름과 가격을 수정하는 함수를 만들고 테스트함.  
Enter만 누르면 기존값을 유지함.  
파일명: `step6_update.py`

```python
# step6_update.py - 사용자 입력으로 데이터 수정
import mysql.connector

def get_conn():
    return mysql.connector.connect(
        host="localhost",
        user="root",
        password="your_password",
        database="starbuuks",
        charset="utf8mb4"
    )

def show_all(cursor):
    cursor.execute("SELECT * FROM coffees ORDER BY id")
    rows = cursor.fetchall()
    print()
    print(f"  {'ID':<5} {'이름':<12} {'가격':>6}")
    print("  " + "-" * 26)
    for row in rows:
        print(f"  {row[0]:<5} {row[1]:<12} {row[2]:>6}원")
    print()

def update(conn, cursor):
    show_all(cursor)
    print("[UPDATE] 수정할 커피 ID를 입력하세요.")
    id_str = input("  ID: ").strip()

    if not id_str.isdigit():
        print("  올바른 ID를 입력하세요.")
        return

    cursor.execute("SELECT * FROM coffees WHERE id = %s", (int(id_str),))
    row = cursor.fetchone()
    if not row:
        print(f"  ID={id_str} 항목이 없습니다.")
        return

    print(f"  현재: {row[1]} / {row[2]}원")
    new_name  = input(f"  새 이름 (Enter 입력 시 '{row[1]}' 유지): ").strip()
    new_price = input(f"  새 가격 (Enter 입력 시 {row[2]} 유지): ").strip()

    final_name  = new_name            if new_name            else row[1]
    final_price = int(new_price)      if new_price.isdigit() else row[2]

    cursor.execute(
        "UPDATE coffees SET name = %s, price = %s WHERE id = %s",
        (final_name, final_price, int(id_str))
    )
    conn.commit()
    print(f"  ID={id_str} 수정 완료 -> {final_name} / {final_price}원")

# 단독 실행 테스트
conn   = get_conn()
cursor = conn.cursor()

update(conn, cursor)

print("[수정 후] 현재 목록:")
show_all(cursor)

cursor.close()
conn.close()
```

## 11. [Step 7] DELETE - 사용자 입력으로 삭제

ID를 입력받아 해당 행을 삭제하는 함수를 만들고 테스트함.  
삭제 전 확인 절차(y/n)를 거침.  
파일명: `step7_delete.py`

```python
# step7_delete.py - 사용자 입력으로 데이터 삭제
import mysql.connector

def get_conn():
    return mysql.connector.connect(
        host="localhost",
        user="root",
        password="your_password",
        database="starbuuks",
        charset="utf8mb4"
    )

def show_all(cursor):
    cursor.execute("SELECT * FROM coffees ORDER BY id")
    rows = cursor.fetchall()
    print()
    print(f"  {'ID':<5} {'이름':<12} {'가격':>6}")
    print("  " + "-" * 26)
    for row in rows:
        print(f"  {row[0]:<5} {row[1]:<12} {row[2]:>6}원")
    print()

def delete(conn, cursor):
    show_all(cursor)
    print("[DELETE] 삭제할 커피 ID를 입력하세요.")
    id_str = input("  ID: ").strip()

    if not id_str.isdigit():
        print("  올바른 ID를 입력하세요.")
        return

    cursor.execute("SELECT name FROM coffees WHERE id = %s", (int(id_str),))
    row = cursor.fetchone()
    if not row:
        print(f"  ID={id_str} 항목이 없습니다.")
        return

    confirm = input(f"  '{row[0]}'을(를) 정말 삭제하시겠습니까? (y/n): ").strip().lower()
    if confirm != "y":
        print("  삭제 취소")
        return

    cursor.execute("DELETE FROM coffees WHERE id = %s", (int(id_str),))
    conn.commit()
    print(f"  ID={id_str} '{row[0]}' 삭제 완료")

# 단독 실행 테스트

cursor = conn.cursor()

delete(conn, cursor)

print("[삭제 후] 현재 목록:")

cursor.close()

```

## 핵심 개념 요약

| 코드 | 역할 | 비고 |
|---|---|---|
| `CREATE DATABASE IF NOT EXISTS` | DB 없으면 생성, 있으면 유지 | 반복 실행 안전 |
| `CREATE TABLE IF NOT EXISTS` | 테이블 없으면 생성, 있으면 유지 | 반복 실행 안전 |
| `executemany(sql, list)` | 리스트를 한 번에 INSERT | 반복 INSERT보다 효율적 |
| `cursor.fetchone()` | 결과 1행만 반환 | COUNT, 단건 조회에 사용 |
| `input().strip()` | 사용자 입력 + 앞뒤 공백 제거 | 입력값 정제 기본 패턴 |
| `isdigit()` | 숫자 문자열 여부 확인 | 잘못된 입력 방어 |
| `%s` 플레이스홀더(Placeholder) | SQL 파라미터 바인딩(Binding) | SQL 인젝션(Injection) 방지 |

> [Fact] `input()`으로 받은 값을 `f"...{변수}..."` 방식으로 SQL에 직접 넣으면
> SQL 인젝션(Injection) 공격에 취약함. 반드시 `%s` + 튜플로 전달해야 함.
